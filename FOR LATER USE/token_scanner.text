import 'dart:io';
import 'dart:math';

import 'package:flutter/material.dart';
import 'package:google_mlkit_text_recognition/google_mlkit_text_recognition.dart';
import 'package:image_picker/image_picker.dart';

/// One token line read from the image.
class ScannedRow {
  String number;
  String fr;
  String sr;

  ScannedRow({required this.number, this.fr = '', this.sr = ''});
}

// PARSER  (same rules as pasteTokenData on the main screen)
class TokenTextParser {
  TokenTextParser._();

  // First line of the sheet like "06-11" is the date.
  static final RegExp _dateHeader = RegExp(r'^\d{1,2}\s*[-/.]\s*\d{1,2}(?!\d)');

  static final RegExp _whatsappHeader = RegExp(
    r'\[\s*\d{1,2}[/.\-]\d{1,2}(?:[/.\-]\d{2,4})?,?\s*'
    r'\d{1,2}:\d{2}(?::\d{2})?\s*(?:[ap]\.?m\.?)?\s*\]\s*[^:\n\]]*:'
    r'|'
    r'\d{1,2}[/.\-]\d{1,2}[/.\-]\d{2,4},?\s*'
    r'\d{1,2}:\d{2}(?::\d{2})?\s*(?:[ap]\.?m\.?)?\s*[-–]\s*[^:\n]*:',
    caseSensitive: false,
  );

  static final RegExp _dateOrTime = RegExp(
    r'\d{1,2}[/.]\d{1,2}[/.]\d{2,4}'
    r'|\d{1,2}:\d{2}:\d{2}(?:\s*[ap]\.?m\.?)?'
    r'|\d{1,2}:\d{2}\s*[ap]\.?m\.?',
    caseSensitive: false,
  );

  static final RegExp _totalLine = RegExp(
    r'^(?:total|grand\s+total)\b',
    caseSensitive: false,
  );

  // "Total: 50" -> 50
  static final RegExp _printedTotal = RegExp(
    r'^(?:total|grand\s+total)\D*(\d+)',
    caseSensitive: false,
  );

  // Camera watermark such as "REDMI NOTE 10 PRO MAX | SRB".
  static final RegExp _cameraMark = RegExp(
    r'redmi|samsung|\bvivo\b|oppo|realme|poco|iphone|shot\s+on|\bsrb\b|pro\s*max|\bnote\s*\d+',
    caseSensitive: false,
  );

  // Header lines such as "Num  TR" or "Num  FR  SR".
  static final RegExp _headerLine = RegExp(
    r'^(?:num|number|fr|sr|tr|[^a-z0-9])+$',
    caseSensitive: false,
  );

  // Circled total read with brackets, e.g. "(280)".
  static final RegExp _circledTotal = RegExp(r'^[\(\[\{]\s*\d+\s*[\)\]\}]$');

  static final RegExp _lonelyX = RegExp(r'(^|\s)[xX](?=\s|$)');

  static final RegExp _tokenPattern = RegExp(r'\d+|[A-Za-z]+');
  static final RegExp _digitsOnly = RegExp(r'^\d+$');
  static final RegExp _allPair = RegExp(r'^A[AL]{0,3}[PF][AEI]{0,2}RS?$');
  static final RegExp _halfPair = RegExp(
    r'^(?=.*[FP])H[AEL]{0,3}F?P?[AEI]{0,2}RS?$',
  );

  /// Fixes common OCR mistakes.
  static String _fixOcr(String s) {
    var t = s
        .replaceAll('₹', ' ')
        .replaceAll(RegExp(r'\bRs\.?(?=\s|\d)', caseSensitive: false), ' ');

    // A handwritten 5 at the start of "5H 25" is often read as f / s.
    t = t.replaceFirst(RegExp(r'^\s*[fFsS](?=\s?[HhEe]\s?\d)'), '5');

    // A tall handwritten "1" is often read as | I l ! -> "10".
    t = t.replaceAllMapped(
      RegExp(r'(?<![\dA-Za-z])[|Il!]\s?[0Oo](?!\d)'),
      (_) => '10',
    );

    t = t
        .replaceAll(RegExp(r'(?<=\d)[Oo]|[Oo](?=\d)'), '0')
        .replaceAll(RegExp(r'(?<=\d)[lI](?=\d)'), '1')
        .replaceAll(RegExp(r'(?<=\d)[Ss](?![Rr])|[Ss](?=\d)'), '5')
        .replaceAll('|', ' ');

    // "1 0" / "5 0" written with a gap -> 10 / 50.
    t = t.replaceAllMapped(
      RegExp(r'(?<!\d)([1-9])\s0(?!\d)'),
      (m) => '${m[1]}0',
    );

    return t;
  }

  // Letters that look like H / E / V in handwriting.
  static String? _heLetter(String t) {
    if (t == 'FR' || t == 'SR' || t == 'TR') return null;
    final c = t[0];
    if (c == 'H' || c == 'N' || c == 'K' || c == 'M') return 'H';
    if (c == 'E' || c == 'F' || c == 'C') return 'E';
    return null;
  }

  static bool _isVLetter(String t) {
    final c = t[0];
    return c == 'V' || c == 'U' || c == 'Y';
  }

  static List<ScannedRow> parse(String rawText) {
    final text = rawText.replaceAll(_whatsappHeader, '\n');
    final result = <ScannedRow>[];

    var seenFirst = false;
    var receipt = false; // printed receipt with a "Num ..." header
    var receiptTR = false; // receipt has only one amount column (TR)
    int? printedTotal;

    for (final line in text.split(RegExp(r'\r?\n'))) {
      // Camera watermark is never a token.
      if (_cameraMark.hasMatch(line)) continue;

      var cleaned = _fixOcr(line).replaceAll(_dateOrTime, ' ').trim();
      if (cleaned.isEmpty) continue;

      // "Num  TR" / "Num  FR  SR" header.
      if (_headerLine.hasMatch(cleaned)) {
        final up = cleaned.toUpperCase();
        if (up.contains('NUM')) {
          receipt = true;
          receiptTR = up.contains('TR') && !up.contains('FR');
        }
        continue;
      }

      // "Total: 50" line.
      final tm = _printedTotal.firstMatch(cleaned);
      if (tm != null) {
        printedTotal = int.tryParse(tm.group(1)!);
        continue;
      }
      if (_totalLine.hasMatch(cleaned)) continue;

      // Circled total read as "(280)".
      if (_circledTotal.hasMatch(cleaned)) continue;

      // The first line of the sheet is the date (06-11), not a token.
      if (!seenFirst) {
        seenFirst = true;
        if (_dateHeader.hasMatch(cleaned)) continue;
      }

      cleaned = cleaned.replaceAllMapped(_lonelyX, (m) => '${m[1]}XBLANK');

      final tokens =
          _tokenPattern
              .allMatches(cleaned)
              .map((m) => m.group(0)!.toUpperCase())
              .toList();

      if (tokens.isEmpty) continue;

      // Num written with a gap: "5 8" -> "58", "0 7" -> "07".
      if (tokens.length >= 2 &&
          tokens[0].length == 1 &&
          tokens[1].length == 1 &&
          _digitsOnly.hasMatch(tokens[0]) &&
          _digitsOnly.hasMatch(tokens[1])) {
        tokens.replaceRange(0, 2, [tokens[0] + tokens[1]]);
      }

      // ---------- 1. Find the Num ----------
      String number = '';
      int next = 0;
      String? compactAmount;

      final first = tokens[0];

      if (_digitsOnly.hasMatch(first)) {
        final hasNextLetters =
            tokens.length > 1 && !_digitsOnly.hasMatch(tokens[1]);

        if (first.length == 2) {
          number = first;
          next = 1;
          if (hasNextLetters && _isVLetter(tokens[1])) {
            number += 'V';
            next = 2;
          }
        } else if (first.length == 1) {
          final he = hasNextLetters ? _heLetter(tokens[1]) : null;
          if (he != null) {
            number = first + he;
            next = 2;
          } else {
            continue;
          }
        } else {
          final possibleAmount = first.substring(2);
          final amount = int.tryParse(possibleAmount);

          if (amount == null || amount < 1 || amount > 15000) continue;

          number = first.substring(0, 2);
          compactAmount = possibleAmount;
          next = 1;
        }
      } else {
        final second =
            tokens.length > 1 && !_digitsOnly.hasMatch(tokens[1])
                ? tokens[1]
                : '';

        if (first.startsWith('AP') || first.startsWith('HP')) {
          number = first.substring(0, 2);
          next = 1;
        } else if ((first == 'A' || first == 'H') &&
            tokens.length > 1 &&
            tokens[1].startsWith('P')) {
          number = '${first}P';
          next = 2;
        } else if (_allPair.hasMatch(first)) {
          number = 'AP';
          next = 1;
        } else if (_halfPair.hasMatch(first)) {
          number = 'HP';
          next = 1;
        } else if (second.isNotEmpty && _allPair.hasMatch(first + second)) {
          number = 'AP';
          next = 2;
        } else if (second.isNotEmpty && _halfPair.hasMatch(first + second)) {
          number = 'HP';
          next = 2;
        } else {
          continue;
        }
      }

      // ---------- 2. Find the amounts ----------
      String? frLabel, srLabel, trLabel;
      final unlabelled = <String>[];

      if (compactAmount != null) unlabelled.add(compactAmount);

      String? pendingLabel;

      for (int i = next; i < tokens.length; i++) {
        final t = tokens[i];

        // On a receipt, a lone "7" before an amount is the ₹ sign misread.
        if (receipt &&
            t == '7' &&
            i + 1 < tokens.length &&
            _digitsOnly.hasMatch(tokens[i + 1])) {
          continue;
        }

        if (t == 'XBLANK') {
          if (pendingLabel == null) unlabelled.add('');
          pendingLabel = null;
          continue;
        }

        if (_digitsOnly.hasMatch(t)) {
          final amount = int.tryParse(t);
          final valid = amount != null && amount >= 1 && amount <= 15000;

          if (valid) {
            if (pendingLabel == 'FR') {
              frLabel = t;
            } else if (pendingLabel == 'SR') {
              srLabel = t;
            } else if (pendingLabel == 'TR') {
              trLabel = t;
            } else {
              unlabelled.add(t);
            }
          }
          pendingLabel = null;
        } else {
          pendingLabel = (t == 'FR' || t == 'SR' || t == 'TR') ? t : null;
        }
      }

      // ---------- 3. Keep the amounts (all are added together later) ----------
      final filled = unlabelled.where((a) => a.isNotEmpty).toList();
      String fr = frLabel ?? '';
      String sr = srLabel ?? trLabel ?? '';

      if (receiptTR) {
        // TR receipt: exactly one amount per row.
        fr = '';
        sr =
            trLabel ??
            srLabel ??
            frLabel ??
            (filled.isNotEmpty ? filled.last : '');
      } else {
        final rest = List<String>.from(unlabelled);
        if (fr.isEmpty && rest.isNotEmpty) fr = rest.removeAt(0);
        if (sr.isEmpty && rest.isNotEmpty) sr = rest.removeAt(0);
      }

      result.add(ScannedRow(number: number, fr: fr, sr: sr));

      if (result.length >= 200) break;
    }

    if (receipt) return _finishReceipt(result, printedTotal);
    return _finish(result);
  }

  // ---------- post-processing ----------

  static bool _noAmount(ScannedRow r) => r.fr.isEmpty && r.sr.isEmpty;

  static int rowTotal(ScannedRow r) =>
      ((int.tryParse(r.fr) ?? 0) + (int.tryParse(r.sr) ?? 0)) *
      multiplier(r.number);

  static int _sum(List<ScannedRow> rows) =>
      rows.fold<int>(0, (s, r) => s + rowTotal(r));

  // "75" -> "5", "720" -> "20" (the ₹ sign read as a leading 7).
  static String _strip7(String a) {
    if (a.length >= 2 && a.startsWith('7')) {
      final v = int.tryParse(a.substring(1));
      if (v != null && v >= 1) return '$v';
    }
    return a;
  }

  /// Receipt: use the printed "Total" to check the amounts.
  static List<ScannedRow> _finishReceipt(List<ScannedRow> rows, int? printed) {
    if (printed == null || _sum(rows) == printed) return rows;

    final alt =
        rows
            .map(
              (r) => ScannedRow(
                number: r.number,
                fr: _strip7(r.fr),
                sr: _strip7(r.sr),
              ),
            )
            .toList();

    return _sum(alt) == printed ? alt : rows;
  }

  static List<ScannedRow> _finish(List<ScannedRow> rows) {
    // 1. Only one amount written beside a list -> applies to every number.
    final filled = rows.where((r) => !_noAmount(r)).toList();
    if (filled.length == 1 && rows.length >= 3) {
      final src = filled.first;
      final lastBlank = _noAmount(rows.last);
      final targets = lastBlank ? rows.sublist(0, rows.length - 1) : rows;

      for (final r in targets) {
        if (_noAmount(r)) {
          r.fr = src.fr;
          r.sr = src.sr;
        }
      }

      // Last row could be the circled total (e.g. 50) or a real number.
      if (lastBlank) {
        final sum = _sum(targets);
        if (int.tryParse(rows.last.number) != sum) {
          rows.last.fr = src.fr;
          rows.last.sr = src.sr;
        }
      }
    }

    // 2. Circled total read as a number: its value equals the sum above it.
    if (rows.length > 1 && _noAmount(rows.last)) {
      final sum = _sum(rows.sublist(0, rows.length - 1));
      if (int.tryParse(rows.last.number) == sum) rows.removeLast();
    }

    return rows;
  }

  // ---------- number helpers (same rules as the main screen) ----------

  static bool _isHE(String n) {
    return (n.length == 2 &&
            int.tryParse(n[0]) != null &&
            (n[1] == 'H' || n[1] == 'E')) ||
        n == 'AP' ||
        n == 'HP';
  }

  static bool _isV(String n) => RegExp(r'^[0-9]{2}V$').hasMatch(n);

  static int _vCount(String value) {
    final base = value.substring(0, 2);
    final a = int.parse(base[0]);
    final b = int.parse(base[1]);
    final cA = a >= 5 ? a - 5 : a + 5;
    final cB = b >= 5 ? b - 5 : b + 5;

    return <String>{
      '$a$b',
      '$b$a',
      '$cA$cB',
      '$cB$cA',
      '$a$cB',
      '$cB$a',
      '$cA$b',
      '$b$cA',
    }.length;
  }

  /// HE / AP / HP count as 10 numbers, V counts its generated numbers.
  static int multiplier(String number) {
    if (_isHE(number)) return 10;
    if (_isV(number)) return _vCount(number);
    return 1;
  }
}

class TokenScannerScreen extends StatefulWidget {
  const TokenScannerScreen({super.key});

  @override
  State<TokenScannerScreen> createState() => _TokenScannerScreenState();
}

class _TokenScannerScreenState extends State<TokenScannerScreen> {
  static const Color _primary = Color(0xFF2878D0);
  static const Color _pageColor = Color(0xFFF5F6F8);
  static const Color _border = Color(0xFFD5D9E0);

  final ImagePicker _picker = ImagePicker();

  File? _image;
  List<ScannedRow> _rows = [];
  String _readText = ''; // what the scanner read, for checking

  bool _busy = false;
  bool _scanned = false;
  String? _error;
  int _scanId = 0; // lets us ignore a scan that was cancelled with Back

  // How many numbers were found. The numbers are NOT added to the total.
  int get _numberCount => _rows.length;

  // All amounts (FR + SR + TR) added together with their multipliers.
  int get _total =>
      _rows.fold<int>(0, (sum, r) => sum + TokenTextParser.rowTotal(r));

  // Back arrow on a result goes back to the Scan / Upload start screen.
  bool get _canLeave =>
      !(_scanned || _image != null || _busy || _error != null);

  void _reset() {
    setState(() {
      _scanId++;
      _image = null;
      _rows = [];
      _readText = '';
      _busy = false;
      _scanned = false;
      _error = null;
    });
  }

  Future<void> _pick(ImageSource source) async {
    if (_busy) return;

    try {
      final picked = await _picker.pickImage(source: source, maxWidth: 2600);
      if (picked == null) return;

      final file = File(picked.path);
      final myId = ++_scanId;

      setState(() {
        _image = file;
        _busy = true;
        _scanned = false;
        _error = null;
        _rows = [];
        _readText = '';
      });

      final lines = await _recognize(file);

      if (!mounted || myId != _scanId) return;

      final joined = lines.join('\n');

      setState(() {
        _readText = joined;
        _rows = TokenTextParser.parse(joined);
        _busy = false;
        _scanned = true;
      });
    } catch (e) {
      debugPrint('Token scan failed: $e');
      if (!mounted) return;
      setState(() {
        _busy = false;
        _error = 'Could not read the image. Please try again.';
      });
    }
  }

  /// Reads the image and joins text at the same height into one line.
  Future<List<String>> _recognize(File file) async {
    final recognizer = TextRecognizer(script: TextRecognitionScript.latin);

    try {
      final result = await recognizer.processImage(InputImage.fromFile(file));

      final lines = <TextLine>[];
      for (final block in result.blocks) {
        lines.addAll(block.lines);
      }
      if (lines.isEmpty) return [];

      lines.sort(
        (a, b) => a.boundingBox.center.dy.compareTo(b.boundingBox.center.dy),
      );

      final rows = <List<TextLine>>[];
      double rowCenter = 0;
      double rowHeight = 0;

      for (final line in lines) {
        final center = line.boundingBox.center.dy;
        final height = line.boundingBox.height;

        if (rows.isNotEmpty &&
            (center - rowCenter).abs() <= max(rowHeight, height) * 0.6) {
          rows.last.add(line);
          rowCenter =
              rows.last
                  .map((l) => l.boundingBox.center.dy)
                  .reduce((a, b) => a + b) /
              rows.last.length;
          rowHeight = max(rowHeight, height);
        } else {
          rows.add([line]);
          rowCenter = center;
          rowHeight = height;
        }
      }

      return rows.map((row) {
        row.sort((a, b) => a.boundingBox.left.compareTo(b.boundingBox.left));
        return row.map((l) => l.text).join('  ');
      }).toList();
    } finally {
      recognizer.close();
    }
  }

  void _showFullImage() {
    final file = _image;
    if (file == null) return;

    showDialog<void>(
      context: context,
      builder:
          (_) => Dialog(
            insetPadding: const EdgeInsets.all(8),
            child: InteractiveViewer(
              maxScale: 6,
              child: Image.file(file, fit: BoxFit.contain),
            ),
          ),
    );
  }

  Widget _stat(String label, String value, {bool bold = false}) {
    return Expanded(
      child: Column(
        children: [
          Text(
            label,
            style: const TextStyle(fontSize: 13, color: Color(0xFF687282)),
          ),
          const SizedBox(height: 6),
          Text(
            value,
            style: TextStyle(
              fontSize: bold ? 26 : 22,
              fontWeight: FontWeight.bold,
              color: bold ? _primary : Colors.black,
            ),
          ),
        ],
      ),
    );
  }

  Widget _resultCard() {
    return Card(
      color: Colors.white,
      elevation: 0,
      shape: RoundedRectangleBorder(
        borderRadius: BorderRadius.circular(12),
        side: const BorderSide(color: _border),
      ),
      child: Padding(
        padding: const EdgeInsets.symmetric(vertical: 20, horizontal: 8),
        child: Row(
          children: [
            _stat('Numbers', '$_numberCount'),
            _stat('Total', '₹ $_total', bold: true),
          ],
        ),
      ),
    );
  }

  Widget _readTextPanel() {
    return Card(
      color: Colors.white,
      elevation: 0,
      margin: const EdgeInsets.only(top: 10),
      shape: RoundedRectangleBorder(
        borderRadius: BorderRadius.circular(12),
        side: const BorderSide(color: _border),
      ),
      child: Theme(
        data: Theme.of(context).copyWith(dividerColor: Colors.transparent),
        child: ExpansionTile(
          title: const Text(
            'What the scanner read',
            style: TextStyle(fontSize: 14),
          ),
          childrenPadding: const EdgeInsets.fromLTRB(16, 0, 16, 12),
          expandedCrossAxisAlignment: CrossAxisAlignment.start,
          children: [
            SelectableText(
              _readText.isEmpty ? '(nothing read)' : _readText,
              style: const TextStyle(fontSize: 13, color: Color(0xFF444B57)),
            ),
          ],
        ),
      ),
    );
  }

  @override
  Widget build(BuildContext context) {
    final showResults = _scanned && !_busy;

    return PopScope(
      canPop: _canLeave,
      onPopInvokedWithResult: (didPop, result) {
        if (!didPop) _reset();
      },
      child: Scaffold(
        backgroundColor: _pageColor,
        appBar: AppBar(
          title: const Text('Token Scanner'),
          backgroundColor: Colors.white,
          surfaceTintColor: Colors.transparent,
        ),
        body: ListView(
          padding: const EdgeInsets.all(16),
          children: [
            Row(
              children: [
                Expanded(
                  child: ElevatedButton.icon(
                    onPressed: _busy ? null : () => _pick(ImageSource.camera),
                    icon: const Icon(Icons.photo_camera_outlined),
                    label: const Text('Scan'),
                    style: ElevatedButton.styleFrom(
                      backgroundColor: _primary,
                      foregroundColor: Colors.white,
                      minimumSize: const Size(0, 48),
                    ),
                  ),
                ),
                const SizedBox(width: 12),
                Expanded(
                  child: OutlinedButton.icon(
                    onPressed: _busy ? null : () => _pick(ImageSource.gallery),
                    icon: const Icon(Icons.upload_file),
                    label: const Text('Upload'),
                    style: OutlinedButton.styleFrom(
                      foregroundColor: _primary,
                      minimumSize: const Size(0, 48),
                    ),
                  ),
                ),
              ],
            ),

            if (_image != null) ...[
              const SizedBox(height: 12),
              GestureDetector(
                onTap: _showFullImage,
                child: ClipRRect(
                  borderRadius: BorderRadius.circular(12),
                  child: Image.file(
                    _image!,
                    width: double.infinity,
                    fit: BoxFit.fitWidth,
                  ),
                ),
              ),
              const Padding(
                padding: EdgeInsets.only(top: 4),
                child: Text(
                  'Tap the image to zoom',
                  style: TextStyle(fontSize: 11, color: Colors.grey),
                ),
              ),
            ],

            if (_busy)
              const Padding(
                padding: EdgeInsets.symmetric(vertical: 24),
                child: Center(
                  child: Column(
                    children: [
                      CircularProgressIndicator(),
                      SizedBox(height: 12),
                      Text('Reading tokens...'),
                    ],
                  ),
                ),
              ),

            if (_error != null)
              Padding(
                padding: const EdgeInsets.only(top: 12),
                child: Text(_error!, style: const TextStyle(color: Colors.red)),
              ),

            if (!_scanned && !_busy && _error == null)
              const Padding(
                padding: EdgeInsets.only(top: 28),
                child: Text(
                  'Scan a token with the camera or upload a photo. '
                  'The app counts the numbers and adds all amounts together.',
                  textAlign: TextAlign.center,
                  style: TextStyle(color: Color(0xFF687282)),
                ),
              ),

            if (showResults) ...[
              const SizedBox(height: 14),
              if (_rows.isNotEmpty)
                _resultCard()
              else
                const Padding(
                  padding: EdgeInsets.symmetric(vertical: 12),
                  child: Text(
                    'No tokens found. Try a clearer, straight photo.',
                    textAlign: TextAlign.center,
                    style: TextStyle(color: Color(0xFF687282)),
                  ),
                ),
              _readTextPanel(),
              const SizedBox(height: 24),
            ],
          ],
        ),
      ),
    );
  }
}
